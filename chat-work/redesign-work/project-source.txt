app.post('/api/developer-login', (req, res) => {
  const code = String(req.body?.code || '');
  const expected = String(process.env.DEVELOPER_CODE);

  const a = Buffer.from(code);
  const b = Buffer.from(expected);

  if (a.length !== b.length || !crypto.timingSafeEqual(a, b)) {
    return res.status(401).json({ success: false });
  }

  req.session.isDeveloper = true;

  req.session.save(err => {
    if (err) {
      return res.status(500).json({ success: false });
    }

    res.json({ success: true });
  });
});

const crypto = require('crypto');
const fs = require('fs');
const path = require('path');
const multer = require('multer');

module.exports = function installChat(app, options) {
  const { dataDir, requireDeveloper, getStore, saveStore } = options;

  const uploadDir = path.join(dataDir, 'chat-uploads');
  fs.mkdirSync(uploadDir, { recursive: true });

  const storage = multer.diskStorage({
    destination: (req, file, cb) => cb(null, uploadDir),
    filename: (req, file, cb) => {
      const ext = path.extname(file.originalname).slice(0, 12);
      cb(null, crypto.randomUUID() + ext);
    }
  });

  const upload = multer({
    storage,
    limits: { fileSize: 3 * 1024 * 1024 },
    fileFilter: (req, file, cb) => {
      const allowed = [
        'image/jpeg', 'image/png', 'image/webp',
        'image/gif', 'video/mp4', 'video/webm',
        'application/pdf'
      ];
      cb(null, allowed.includes(file.mimetype));
    }
  });

  const store = getStore();
  store.chats ||= [];

  function findChat(id) {
    return store.chats.find(c => c.id === id);
  }

  function guestAuth(req, res, next) {
    const chat = findChat(req.params.id);
    const token = String(req.headers['x-chat-token'] || req.query.token || '');
    if (!chat || !token || !chat.tokenHash) {
      return res.status(401).json({ error: 'المحادثة غير متاحة' });
    }

    const hash = crypto.createHash('sha256').update(token).digest();
    const saved = Buffer.from(chat.tokenHash, 'hex');

    if (saved.length !== hash.length || !crypto.timingSafeEqual(saved, hash)) {
      return res.status(401).json({ error: 'بيانات المحادثة غير صحيحة' });
    }
    req.chat = chat;
    next();
  }

  function cleanMessage(m) {
    return {
      id: m.id,
      sender: m.sender,
      text: m.text || '',
      file: m.file || null,
      date: m.date
    };
  }

  app.post('/api/chat/start', (req, res) => {
    const name = String(req.body?.name || '').trim().slice(0, 60);
    const phone = String(req.body?.phone || '').trim().slice(0, 25);

    if (!name) return res.status(400).json({ error: 'اكتب اسمك أولاً' });

    const token = crypto.randomBytes(32).toString('hex');
    const chat = {
      id: crypto.randomUUID(),
      name,
      phone,
      tokenHash: crypto.createHash('sha256').update(token).digest('hex'),
      createdAt: new Date().toISOString(),
      messages: []
    };

    store.chats.push(chat);
    saveStore();

    res.json({ success: true, id: chat.id, token });
  });

  app.get('/api/chat/:id', guestAuth, (req, res) => {
    res.json({
      chat: {
        id: req.chat.id,
        name: req.chat.name,
        messages: req.chat.messages.map(cleanMessage)
      }
    });
  });

  app.post('/api/chat/:id/messages', guestAuth, upload.single('file'), (req, res) => {
    const text = String(req.body?.text || '').trim().slice(0, 3000);
    if (!text && !req.file) {
      return res.status(400).json({ error: 'اكتب رسالة أو اختر ملفًا' });
    }

    const message = {
      id: crypto.randomUUID(),
      sender: 'visitor',
      text,
      date: new Date().toISOString(),
      file: req.file ? {
        name: req.file.originalname.slice(0, 120),
        type: req.file.mimetype,
        path: req.file.filename,
        size: req.file.size
      } : null
    };

    req.chat.messages.push(message);
    saveStore();
    res.json({ success: true, message: cleanMessage(message) });
  });

  app.get('/api/admin/chats', requireDeveloper, (req, res) => {
    res.json({
      chats: store.chats.map(c => ({
        id: c.id,
        name: c.name,
        phone: c.phone,
        createdAt: c.createdAt,
        lastMessage: c.messages[c.messages.length - 1] || null
      })).reverse()
    });
  });

  app.get('/api/admin/chats/:id', requireDeveloper, (req, res) => {
    const chat = findChat(req.params.id);
    if (!chat) return res.status(404).json({ error: 'المحادثة غير موجودة' });
    res.json({
      chat: {
        id: chat.id,
        name: chat.name,
        phone: chat.phone,
        messages: chat.messages.map(cleanMessage)
      }
    });
  });

  app.post('/api/admin/chats/:id/messages', requireDeveloper, upload.single('file'), (req, res) => {
    const chat = findChat(req.params.id);
    if (!chat) return res.status(404).json({ error: 'المحادثة غير موجودة' });

    const text = String(req.body?.text || '').trim().slice(0, 3000);
    if (!text && !req.file) {
      return res.status(400).json({ error: 'اكتب رسالة أو اختر ملفًا' });
    }

    const message = {
      id: crypto.randomUUID(),
      sender: 'developer',
      text,
      date: new Date().toISOString(),
      file: req.file ? {
        name: req.file.originalname.slice(0, 120),
        type: req.file.mimetype,
        path: req.file.filename,
        size: req.file.size
      } : null
    };

    chat.messages.push(message);
    saveStore();
    res.json({ success: true, message: cleanMessage(message) });
  });

  app.get('/api/chat-media/:id/:messageId', (req, res, next) => {
    const chat = findChat(req.params.id);
    if (!chat) return res.sendStatus(404);

    const isDeveloper = req.session?.isDeveloper === true;
    const token = String(req.headers['x-chat-token'] || req.query.token || '');
    let authorized = isDeveloper;

    if (!authorized && token && chat.tokenHash) {
      const a = crypto.createHash('sha256').update(token).digest();
      const b = Buffer.from(chat.tokenHash, 'hex');
      authorized = a.length === b.length && crypto.timingSafeEqual(a, b);
    }

    if (!authorized) return res.sendStatus(401);

    const msg = chat.messages.find(m => m.id === req.params.messageId);
    if (!msg?.file?.path) return res.sendStatus(404);

    const filePath = path.join(uploadDir, path.basename(msg.file.path));
    if (!fs.existsSync(filePath)) return res.sendStatus(404);

    res.type(msg.file.type);
    res.sendFile(filePath);
  });

  app.delete('/api/admin/chats/:id/messages/:messageId', requireDeveloper, (req, res) => {
    const chat = findChat(req.params.id);
    if (!chat) return res.sendStatus(404);

    const index = chat.messages.findIndex(m => m.id === req.params.messageId);
    if (index < 0) return res.sendStatus(404);

    const [msg] = chat.messages.splice(index, 1);
    if (msg.file?.path) {
      fs.rmSync(path.join(uploadDir, path.basename(msg.file.path)), { force: true });
    }
    saveStore();
    res.json({ success: true });
  });

  app.delete('/api/admin/chats/:id', requireDeveloper, (req, res) => {
    const index = store.chats.findIndex(c => c.id === req.params.id);
    if (index < 0) return res.sendStatus(404);

    const [chat] = store.chats.splice(index, 1);
    for (const msg of chat.messages) {
      if (msg.file?.path) {
        fs.rmSync(path.join(uploadDir, path.basename(msg.file.path)), { force: true });
      }
    }
    saveStore();
    res.json({ success: true });
  });

  app.use((err, req, res, next) => {
    if (err instanceof multer.MulterError) {
      return res.status(400).json({ error: 'حجم الملف الأقصى 3 ميجابايت أو الملف غير صالح' });
    }
    next(err);
  });
};

<script>
(() => {
  const root = document.getElementById('alkfory-search-rating');
  const input = root.querySelector('#alkforySearchInput');
  const results = root.querySelector('#alkforySearchResults');

  function search() {
    const query = input.value.trim().toLocaleLowerCase();
    results.innerHTML = '';

    if (!query) {
      results.style.display = 'none';
      return;
    }

    const elements = document.querySelectorAll(
      'main h1, main h2, main h3, main p, main a, main button, main [data-searchable]'
    );

    const found = [];
    elements.forEach(el => {
      if (root.contains(el)) return;
      const text = (el.innerText || el.textContent || '').trim();
      if (!text || !text.toLocaleLowerCase().includes(query)) return;
      if (found.some(item => item.el === el)) return;
      found.push({el, text});
    });

    if (!found.length) {
      results.innerHTML = '<div style="padding:14px;color:#bbb;text-align:center">لا توجد نتائج مطابقة</div>';
    } else {
      found.slice(0, 30).forEach(({el, text}) => {
        const button = document.createElement('button');
        button.type = 'button';
        button.textContent = text.slice(0, 100);
        button.style.cssText =
          'display:block;width:100%;padding:13px;text-align:right;background:transparent;color:white;border:0;border-bottom:1px solid #333;cursor:pointer;font-size:15px';

        button.addEventListener('click', () => {
          el.scrollIntoView({behavior:'smooth', block:'center'});
          const old = el.style.outline;
          el.style.outline = '2px solid #ed354d';
          setTimeout(() => { el.style.outline = old; }, 1800);
          results.style.display = 'none';
        });

        results.appendChild(button);
      });
    }

    results.style.display = 'block';
  }

  input.addEventListener('input', search);
  input.addEventListener('focus', search);

  document.addEventListener('click', event => {
    if (!root.contains(event.target)) results.style.display = 'none';
  });
})();
</script>

async function askAI() {
  const input = document.getElementById('aiInput');
  const answer = document.getElementById('aiAnswer');
  const message = input.value.trim();
  if (!message) {
    answer.style.display = 'block';
    answer.textContent = english ? 'Please type a question first.' : 'اكتب سؤالك الأول.';
    return;
  }
  answer.style.display = 'block';
  answer.textContent = english ? 'Connecting...' : 'جاري الاتصال...';
  try {
    const response = await fetch('/api/ai', {
      method:'POST',
      headers:{'Content-Type':'application/json'},
      body:JSON.stringify({message})
    });
    const data = await response.json();
    if (!response.ok) throw new Error(data.error || 'AI request failed');
    answer.textContent = data.answer || (english ? 'No response.' : 'لم يصل رد.');
  } catch (error) {
    answer.textContent = (english ? 'Connection error: ' : 'خطأ في الاتصال: ') + error.message;
  }
}

async function loadPublicApps() {
  try {
    const response = await fetch('/api/public/apps');
    if (!response.ok) throw new Error('Apps unavailable');
    const data = await response.json();
    const apps = Array.isArray(data.apps) ? data.apps : [];
    const section = document.getElementById('dynamicApps');
    const list = document.getElementById('publicAppsList');
    if (!section || !list) return;

    list.replaceChildren();

    for (const item of apps) {
      if (!item.name || !item.url) continue;
      let url;
      try {
        url = new URL(item.url);
        if (url.protocol !== 'https:') continue;
      } catch { continue; }

      const card = document.createElement('article');
      card.className = 'card';

      if (item.icon) {
        try {
          const iconUrl = new URL(item.icon);
          if (iconUrl.protocol === 'https:') {
            const icon = document.createElement('img');
            icon.src = iconUrl.href;
            icon.alt = '';
            icon.loading = 'lazy';
            icon.style.cssText = 'width:55px;height:55px;object-fit:contain';
            card.append(icon);
          }
        } catch {}
      }

      const title = document.createElement('h3');
      title.textContent = item.name;
      card.append(title);

      if (item.category) {
        const category = document.createElement('small');
        category.textContent = item.category;
        card.append(category);
      }

      if (item.description) {
        const description = document.createElement('p');
        description.textContent = item.description;
        card.append(description);
      }

      const link = document.createElement('a');
      link.className = 'btn';
      link.textContent = 'فتح التطبيق ↗';
      link.href = url.href;
      link.target = '_blank';
      link.rel = 'noopener noreferrer';
      card.append(link);
      list.append(card);
    }

    section.style.display = apps.length ? '' : 'none';
  } catch (error) {
    console.warn('تعذر تحميل التطبيقات:', error.message);
  }
}

async function sendVisitorMessage() {

  }

  function goToChat(e) {
    const el = e.target.closest('a,button,[role="button"]');

  function fixCopyright() {
    // حذف n أو \n إذا كانا نصًا منفصلًا في نهاية الصفحة فقط
    const walker = document.createTreeWalker(
      document.body, NodeFilter.SHOW_TEXT
    );
    const nodes = [];
    while (walker.nextNode()) nodes.push(walker.currentNode);

    for (const node of nodes) {
      const value = node.nodeValue.trim();
      if (value === 'n' || value === '\\n') {
        const parent = node.parentElement;
        if (parent && !parent.closest('script,style,textarea,input')) {
          node.remove();
        }
      }
    }

    // نقل حقوق النشر إلى آخر الصفحة
    const candidates = [...document.body.querySelectorAll('p,small,div,span')]
      .filter(el => {
        const t = (el.innerText || '').trim();
        return t.length < 180 &&
          (t.includes('حقوق النشر') ||
           t.includes('جميع الحقوق محفوظة') ||
           t.includes('All rights reserved'));
      });

    let copyright = candidates.find(el =>
      ![...el.children].some(child =>
        (child.innerText || '').includes('حقوق النشر')
      )
    ) || candidates[0];

    if (copyright) {
      copyright.id = 'alkforyCopyright';
      const footer = document.querySelector('footer') ||
        document.body.appendChild(document.createElement('footer'));
      footer.appendChild(copyright);
    }
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', fixCopyright);
  } else {
    fixCopyright();
  }
})();

function readFields(keys){
 const out={};
 for(const id of keys){
  const el=document.getElementById(id);
  if(!el)continue;
  if(el.tagName==='SELECT' && ['true','false'].includes(el.value))out[id]=el.value==='true';
  else if(el.type==='number')out[id]=el.value===''?'':Number(el.value);
  else out[id]=el.value;
 }
 return out;
}

self.addEventListener('message', event => {
  if (event.data && event.data.type === 'SKIP_WAITING') {
    self.skipWaiting();
  }
});

const CACHE_NAME = 'alkfory-shell-v4';
const APP_SHELL = [
  '/',
  '/entry.html',
  '/manifest.json',
  '/icons/alkfory.svg',
  '/audio/entry-1.mp3',
  '/audio/entry-2.mp3',
  '/audio/profile-recording.mp3'
];

self.addEventListener('install', event => {
  event.waitUntil(
    caches.open(CACHE_NAME)
      .then(cache => cache.addAll(APP_SHELL))
      .then(() => self.skipWaiting())
  );
});

self.addEventListener('activate', event => {
  event.waitUntil(
    caches.keys().then(keys =>
      Promise.all(keys
        .filter(key => key.startsWith('alkfory-shell-') && key !== CACHE_NAME)
        .map(key => caches.delete(key))
      )
    ).then(() => self.clients.claim())
  );
});

self.addEventListener('fetch', event => {
  const request = event.request;
  const url = new URL(request.url);

  if (request.method !== 'GET' || url.origin !== self.location.origin) return;

  // لا نخزّن API أو صفحات خاصة أو ردود تسجيل الدخول.
  if (url.pathname.startsWith('/api/') ||
      url.pathname === '/admin' ||
      url.pathname === '/home.html' ||
      url.pathname.startsWith('/api')) return;

  if (request.mode === 'navigate') {
    event.respondWith(
      fetch(request).catch(() =>
        caches.match('/entry.html')
      )
    );
  }
});
